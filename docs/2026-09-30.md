### 추상화와 인터페이스

## 서론

# 추상화
    - 추상화는 어떠한 클래스의 재료들을 만들때에 사용하는 클래스다. 동시에 어떠한 메서드나 기입 값이 미정일 때 사용할 수 있다. 추상화는 완성의 책임이 존재하지 않는다. 그렇기에 다수의 추상 메서드를 가질 수 있다.

# 인터페이스
    - 인터페이스는 필드와 추상 메서드를 가질 수 있다. 그런데 인터페이스에서 선언되는 필드값은 자동으로 public static final로 지정되기에 선언 값을 지정해야한다. 인터페이스는 다수의 기능을 묶어서 표현할때 사용한다. 인터페이스는 대체로 can do~ 형식으로 이해될 수 있는데 -is a 도 가능하지만 보통은 can do.- 상속과는 다르게 인터페이스는 다중 구현이 가능하다.

- 인터페이스
// Flyable.java

public interface Flyable {

    void fly();
    // → 메서드 내용이 없는 약속
    // → Flyable을 구현하는 클래스는 fly() 기능을 반드시 만들어야 한다.

    default void land() {
        // → default 메서드는 인터페이스 안에 이미 기능이 있다.
        // → 구현 클래스가 land()를 다시 작성하지 않아도 사용할 수 있다.
        System.out.println("착륙합니다.");  // 해당 부분은 ai가 추가해준 지식이다. 
    }
}

- 추상화 클래스
// TangibleAsset.java

public abstract class TangibleAsset {

    // 모든 유형 자산이 공통으로 가지는 이름
    // → 추상 클래스는 공통 변수(필드)를 가질 수 있다.
    private String name; 

    // 생성자
    // → 추상 클래스는 직접 객체를 만들 수는 없지만,
    //   자식 객체가 만들어질 때 공통 데이터를 초기화할 수 있다.
    public TangibleAsset(String name) {
        this.name = name;
    }

    // 이미 공통으로 사용할 수 있는 메서드
    // → 자식 클래스가 그대로 사용해도 된다.
    public String getName() {
        return name;
    }

    // 내용이 없는 추상 메서드
    // → “모든 자산은 가격을 알려야 한다”는 규칙만 정한다.
    // → 실제 가격 계산 방식은 자식마다 다르므로 자식에게 맡긴다.
    public abstract int price();

    // 내용이 없는 추상 메서드
    // → 모든 자산에는 무게가 있지만, 저장 방식은 자식이 결정한다.
    public abstract double getWeight();
}

- 일반 클래스
// Computer.java

public class Computer extends TangibleAsset {
    // → extends: Computer가 TangibleAsset의 특징을 물려받는다.

    private String makerName;
    // → Computer만 따로 가지는 데이터

    private int computerPrice;
    private double weight;

    public Computer(String name, String makerName, int computerPrice, double weight) {
        super(name);
        // → 부모(TangibleAsset)의 생성자를 호출한다.
        // → 부모에 name이 있으므로 부모가 name을 초기화하게 한다.

        this.makerName = makerName;
        this.computerPrice = computerPrice;
        this.weight = weight;
    }

    public String getMakerName() {
        return makerName;
    }

    public void setMakerName(String makerName) {
        this.makerName = makerName;
    }

    @Override
    public int price() {
        // → TangibleAsset의 abstract int price()를 실제로 구현한다.
        // → Computer가 일반 클래스이므로 이 구현은 반드시 필요하다.
        return computerPrice;
    }

    @Override
    public double getWeight() {
        // → 부모의 추상 메서드를 실제 기능으로 완성한다.
        return weight;
    }

    public void setWeight(double weight) {
        // → Computer에 필요한 일반 메서드는 자유롭게 추가할 수 있다.
        this.weight = weight;
    }
}

- 인터페이스
public interface Flyable {

    // "나는 날 수 있어요"라는 약속만 적어 둔 메서드입니다.
    // 아직 어떻게 나는지는 정하지 않았습니다.
    void fly();

    // default가 붙으면 인터페이스 안에서도
    // 미리 동작하는 메서드를 만들어 둘 수 있습니다.
    // 그래서 Flyable을 사용하는 클래스가 land()를 꼭 다시 만들 필요는 없습니다.
    default void land() {
        System.out.println("착륙합니다.");
    }
}

- 추상화 클래스
public abstract class TangibleAsset {

    // 모든 자산이 공통으로 가질 이름입니다.
    // 예: 컴퓨터 이름, 책 이름, 자동차 이름
    private String name;

    // 자식 클래스가 만들어질 때 name을 함께 저장하기 위한 생성자입니다.
    // 추상 클래스는 직접 new로 만들 수 없지만,
    // 자식 클래스를 만들 때 이 생성자를 사용할 수 있습니다.
    public TangibleAsset(String name) {
        this.name = name;
    }

    // name은 모든 자산에서 같은 방식으로 가져올 수 있으므로
    // 부모 클래스에서 미리 만들어 둡니다.
    public String getName() {
        return name;
    }

    // 모든 자산에는 가격이 있어야 합니다.
    // 하지만 가격을 정하는 방법은 컴퓨터, 책, 자동차마다 다를 수 있습니다.
    // 그래서 여기서는 "price()가 있어야 한다"까지만 정합니다.
    public abstract int price();

    // 모든 자산에는 무게가 있어야 합니다.
    // 실제 무게를 저장하고 반환하는 방법은 자식 클래스가 작성합니다.
    public abstract double getWeight();
}

- 일반 클래스
public class Computer extends TangibleAsset {

    // extends는 "TangibleAsset의 공통 특징을 물려받겠다"는 뜻입니다.
    // 그래서 Computer는 name과 getName()을 사용할 수 있습니다.

    // Computer에만 필요한 정보입니다.
    private String makerName;

    // Computer의 가격과 무게를 저장할 변수입니다.
    private int computerPrice;
    private double weight;

    public Computer(String name, String makerName, int computerPrice, double weight) {

        // 부모 클래스인 TangibleAsset에 name을 전달합니다.
        // 그러면 부모가 name을 저장합니다.
        super(name);

        // Computer에만 있는 정보는 Computer 안에서 저장합니다.
        this.makerName = makerName;
        this.computerPrice = computerPrice;
        this.weight = weight;
    }

    public String getMakerName() {
        return makerName;
    }

    public void setMakerName(String makerName) {
        this.makerName = makerName;
    }

    @Override
    public int price() {

        // 부모가 "price()는 꼭 있어야 해"라고 정해 두었기 때문에
        // Computer는 가격을 반환하는 방법을 실제로 작성해야 합니다.
        return computerPrice;
    }

    @Override
    public double getWeight() {

        // 부모가 정해 둔 getWeight()를
        // Computer의 weight 변수와 연결해서 실제로 작성합니다.
        return weight;
    }

    public void setWeight(double weight) {

        // 이 메서드는 Computer에 필요한 기능이라서
        // 부모에 없어도 자유롭게 추가할 수 있습니다.
        this.weight = weight;
    }
}
